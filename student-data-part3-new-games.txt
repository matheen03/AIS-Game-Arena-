checked as (
  select i.*,
    case
      when exists (select 1 from public.games g where lower(rtrim(g.url, '/')) = lower(rtrim(i.url, '/'))) then 'already in the arena (same link) – kept as it is'
      when exists (select 1 from public.games g where g.cls = i.cls and public._norm_title(g.title) = public._norm_title(i.title)) then 'already in the arena (same name, same class) – kept as it is'
      when exists (select 1 from public.games g, jsonb_array_elements(i.students::jsonb) s
                   where s->>'s' <> '' and g.students @> jsonb_build_array(jsonb_build_object('s', s->>'s'))
                     and (public._norm_title(g.title) like '%' || public._norm_title(i.title) || '%'
                          or public._norm_title(i.title) like '%' || public._norm_title(g.title) || '%'))
           then 'already in the arena (same student, similar name) – kept as it is'
      when exists (select 1 from public.games g where g.id = i.id) then 'skipped (id already used)'
      else 'added'
    end as result
  from incoming i
),
added as (
  insert into public.games (id, cls, title, team, descr, url, students, created_at)
  select id, cls, title, team, descr, url, students::jsonb, created_at::timestamptz from checked where result = 'added'
  on conflict do nothing
  returning id
)
select c.cls as class, c.title as game, c.result
from checked c order by c.result, c.cls, c.title;
